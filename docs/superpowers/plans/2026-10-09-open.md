# 开放平台模块实施计划（P2 第 3 项）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新建 `ccloud-module-open` 服务（端口 28084），交付开放应用、接口目录、应用接口授权、调用日志、回调通知；网关改为经 RPC 从 open 读取应用并做 IP 白名单与接口授权；member 提供第一个开放业务接口「按手机号查询会员信息」。

**Architecture:** open 的结构与 infra 一致（`open-api` 放错误码、常量与 Feign 接口，`open-biz` 为可执行应用），五个框架契约经 RPC 由 system 实现。网关以 WebClient 调 `GET /rpc-api/open/app/get-by-key` 取应用快照，Caffeine 本地缓存，open 变更后经 Redis 发布订阅失效；验签通过后以「开放应用」用户类型写入带签名的内部身份头。业务服务的开放接口调用日志由各服务的 `ApiLogFrameworkServiceImpl` 经 Redis Stream 主题 `open.api-log` 发给 open 写库。回调由 open 用 Apache HttpClient 5 发送，自定义 DNS 解析器拦截内网地址，失败按退避重试。

**Tech Stack:** JDK 21、Spring Boot 3.5.16、Spring Cloud Gateway（WebFlux）、MyBatis Plus 3.5.17、MySQL 8.4、Redis 8；Apache HttpClient 5.5.2（由 Spring Boot 管理版本，本机仓库已有）；测试用 JUnit 5 + AssertJ + MockMvc / WebTestClient，连接本机已有的 MySQL（`ccloud_test` 库）与 Redis（15 号库）；回调对端用 JDK 自带的 `com.sun.net.httpserver.HttpServer` 模拟。

**Spec:** `docs/superpowers/specs/2026-10-09-open-design.md`

```java
package com.ccloud.framework.common.util.net;

import java.net.Inet6Address;
import java.net.InetAddress;
import java.net.UnknownHostException;
import java.util.regex.Pattern;

/**
 * IP 白名单：逗号分隔的 IPv4 / IPv6 地址或 CIDR。只接受字面量地址，绝不做 DNS 解析
 * （域名白名单会让匹配结果随 DNS 变化，解析本身也会阻塞网关的事件循环线程）
 */
public final class IpWhitelistUtils {

    private static final Pattern LITERAL = Pattern.compile("[0-9a-fA-F:.]+");

    private IpWhitelistUtils() {
    }

    public static void validate(String whitelist) {
        if (whitelist == null || whitelist.isBlank()) {
            return;
        }
        for (String entry : whitelist.split(",")) {
            String trimmed = entry.trim();
            if (parse(trimmed) == null) {
                throw new IllegalArgumentException(trimmed);
            }
        }
    }

    public static boolean matches(String whitelist, String ip) {
        if (whitelist == null || whitelist.isBlank()) {
            return true;
        }
        byte[] target = toBytes(ip);
        if (target == null) {
            return false;
        }
        for (String entry : whitelist.split(",")) {
            Range range = parse(entry.trim());
            if (range != null && range.contains(target)) {
                return true;
            }
        }
        return false;
    }

    private record Range(byte[] network, int prefix) {
        boolean contains(byte[] address) {
            if (address.length != network.length) {
                return false;
            }
            int full = prefix / 8;
            for (int i = 0; i < full; i++) {
                if (address[i] != network[i]) {
                    return false;
                }
            }
            int rest = prefix % 8;
            if (rest == 0) {
                return true;
            }
            int mask = (0xFF << (8 - rest)) & 0xFF;
            return (address[full] & mask) == (network[full] & mask);
        }
    }

    private static Range parse(String entry) {
        if (entry.isEmpty()) {
            return null;
        }
        String[] parts = entry.split("/", -1);
        if (parts.length > 2) {
            return null;
        }
        byte[] network = toBytes(parts[0]);
        if (network == null) {
            return null;
        }
        int max = network.length * 8;
        int prefix = max;
        if (parts.length == 2) {
            try {
                prefix = Integer.parseInt(parts[1]);
            } catch (NumberFormatException e) {
                return null;
            }
            if (prefix < 0 || prefix > max) {
                return null;
            }
        }
        return new Range(network, prefix);
    }

    /**
     * 字面量地址转字节；IPv4 映射的 IPv6 地址（::ffff:a.b.c.d）按 IPv4 处理
     */
    private static byte[] toBytes(String ip) {
        if (ip == null || !LITERAL.matcher(ip).matches()) {
            return null;
        }
        try {
            InetAddress address = InetAddress.getByName(ip);   // 已限定为字面量，不会触发 DNS
            byte[] bytes = address.getAddress();
            if (address instanceof Inet6Address && isIpv4Mapped(bytes)) {
                return new byte[]{bytes[12], bytes[13], bytes[14], bytes[15]};
            }
            return bytes;
        } catch (UnknownHostException e) {
            return null;
        }
    }

    private static boolean isIpv4Mapped(byte[] b) {
        for (int i = 0; i < 10; i++) {
            if (b[i] != 0) {
                return false;
            }
        }
        return b[10] == (byte) 0xFF && b[11] == (byte) 0xFF;
    }
}
```

注意：`InetAddress.getByName("::ffff:10.0.0.1")` 在 JDK 中直接返回 `Inet4Address`，上面的映射分支是兜底；测试覆盖两种情况即可。

```java
    private final ObjectProvider<TenantProperties> tenantPropertiesProvider;

    public SystemTenantValidator(ObjectProvider<TenantFrameworkService> tenantFrameworkServiceProvider,
                                 ObjectProvider<TenantProperties> tenantPropertiesProvider) {
        this.tenantFrameworkServiceProvider = tenantFrameworkServiceProvider;
        this.tenantPropertiesProvider = tenantPropertiesProvider;

    public void validate(ErrorCode errorCode) {

        if (properties == null || !Boolean.TRUE.equals(properties.getEnable())) {
            return;

        Long tenantId = TenantContextHolder.getRequiredTenantId();
        if (!tenantFrameworkServiceProvider.getObject().isSystemTenant(tenantId)) {
            throw exception(errorCode);
        }
    }
```

```java
class OpenAppCacheInvalidatorTest extends BaseOpenDbAndRedisUnitTest {

    @Resource
    private OpenAppCacheInvalidator invalidator;
    @Resource
    private StringRedisTemplate stringRedisTemplate;
    @Resource
    private RedisConnectionFactory connectionFactory;
    @Resource
    private PlatformTransactionManager transactionManager;

    @Test
    void 非事务中立即发布_事务中提交后才发布_回滚不发布() throws Exception {
        BlockingQueue<String> received = subscribe();
        invalidator.publish("k1");
        assertThat(received.poll(3, TimeUnit.SECONDS)).isEqualTo("k1");

        TransactionTemplate tx = new TransactionTemplate(transactionManager);
        tx.setPropagationBehavior(TransactionDefinition.PROPAGATION_REQUIRES_NEW);
        tx.executeWithoutResult(status -> {
            invalidator.publish("k2");
            assertThat(received.poll()).isNull();   // 提交前没有发布
        });
        assertThat(received.poll(3, TimeUnit.SECONDS)).isEqualTo("k2");

        tx.executeWithoutResult(status -> {
            invalidator.publishAll();
            status.setRollbackOnly();
        });
        assertThat(received.poll(1, TimeUnit.SECONDS)).isNull();

        invalidator.publishAll();
        assertThat(received.poll(3, TimeUnit.SECONDS)).isEmpty();
    }

    private BlockingQueue<String> subscribe() throws InterruptedException {
```

```java
/**
 * 通知各网关实例清理应用快照缓存。处于事务中时推迟到提交之后：提交前广播，网关可能立即重新加载到旧数据并再次缓存
```

```java
        validateExists(reqVO.getId());

     * 路径只允许字母数字与 _ - / .，且不含 ..：网关按原始路径精确匹配，查询串、通配符、编码字符都会让授权判断与实际路由不一致

    private static OpenApiDO normalize(OpenApiDO api) {
        api.setMethod(api.getMethod().toUpperCase(Locale.ROOT));
        String path = api.getPath();
        if (!PATH.matcher(path).matches() || path.contains("..") || path.contains("//")) {
            throw exception(OPEN_API_PATH_INVALID);
        }
        return api;
    }

    private void validateUnique(Long id, OpenApiDO api) {
        OpenApiDO byCode = openApiMapper.selectByCode(api.getCode());
        if (byCode != null && !byCode.getId().equals(id)) {
            throw exception(OPEN_API_CODE_EXISTS);
        }
        OpenApiDO byPath = openApiMapper.selectByMethodAndPath(api.getMethod(), api.getPath());
        if (byPath != null && !byPath.getId().equals(id)) {
            throw exception(OPEN_API_METHOD_PATH_EXISTS);
```

```java
package com.ccloud.module.open.service.app;

/**
 * 应用标识与密钥的生成、掩码。用 SecureRandom：密钥可被预测就等于可被伪造签名
 */
public final class OpenAppSecretUtils {

    private static final SecureRandom RANDOM = new SecureRandom();
    private static final String LOWER_DIGITS = "abcdefghijklmnopqrstuvwxyz0123456789";
    private static final String ALNUM = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";

    private OpenAppSecretUtils() {
    }

    public static String generateAppKey() {
        return random(LOWER_DIGITS, 16);
    }

    public static String generateSecret() {
        return random(ALNUM, 32);
    }

    /**
     * 列表、详情中展示的密钥：前 4 位 + **** + 后 4 位；长度不足 12 位时整体隐藏
     */
    public static String mask(String secret) {
        if (secret == null) {
            return null;
        }
        if (secret.length() < 12) {
            return "****";
        }
        return secret.substring(0, 4) + "****" + secret.substring(secret.length() - 4);
    }

    private static String random(String chars, int length) {
        StringBuilder sb = new StringBuilder(length);
        for (int i = 0; i < length; i++) {
            sb.append(chars.charAt(RANDOM.nextInt(chars.length())));
        }
        return sb.toString();
    }
}
```

```java
@Slf4j
@Service
public class OpenAppServiceImpl implements OpenAppService {

    static final String GET_URL = "http://ccloud-open-server/rpc-api/open/app/get-by-key?appKey={appKey}";

    private final WebClient webClient;
    private final OpenApiProperties properties;
    private final AsyncCache<String, Optional<OpenApp>> cache;

    public OpenAppServiceImpl(WebClient.Builder loadBalancedWebClientBuilder, OpenApiProperties properties) {
        this.webClient = loadBalancedWebClientBuilder.build();
        this.properties = properties;
        // 存在的应用缓存较久；不存在的短暂缓存，防止随机 appKey 打穿到 open，又能让新建的应用尽快可用
        this.cache = Caffeine.newBuilder()
                .expireAfter(Expiry.creating((String key, Optional<OpenApp> value) ->
                        value.isPresent() ? properties.getAppCacheTtl() : properties.getAppCacheNotFoundTtl()))
                .maximumSize(properties.getAppCacheMaxSize())
                .buildAsync();
    }

    @Override
    public Mono<OpenApp> getApp(String appKey) {
        return Mono.defer(() -> {
            CompletableFuture<Optional<OpenApp>> future =
                    cache.get(appKey, (key, executor) -> remoteGet(key).toFuture());
            return Mono.fromFuture(future, true).doOnError(e -> cache.asMap().remove(appKey, future));
        }).flatMap(Mono::justOrEmpty);
    }

    private Mono<Optional<OpenApp>> remoteGet(String appKey) {
        return webClient.get().uri(GET_URL, appKey)
                .retrieve()
                .bodyToMono(String.class)
                .timeout(properties.getAppLookupTimeout())
                .onErrorMap(e -> new OpenAppUnavailableException("开放平台服务不可用", e))
                .map(OpenAppServiceImpl::parse);
    }

    private static Optional<OpenApp> parse(String body) {
        CommonResult<OpenApp> result;
        try {
            result = JsonUtils.parseObject(body, new TypeReference<>() {});
        } catch (RuntimeException e) {
            throw new OpenAppUnavailableException("应用查询响应无法解析", e);
        }
        if (result == null || !result.isSuccess()) {
            throw new OpenAppUnavailableException("应用查询返回错误码 " + (result == null ? null : result.getCode()), null);
        }
        return Optional.ofNullable(result.getData());
    }

    @Override
    public void invalidate(String appKey) {
        cache.synchronous().invalidate(appKey);
    }

    @Override
    public void invalidateAll() {
        cache.synchronous().invalidateAll();
    }
}
```

```java
@Slf4j
@Component
public class OpenAppInvalidationListener implements SmartLifecycle {

    public static final String CHANNEL = "ccloud:open:app:invalidate";

    private final ReactiveRedisConnectionFactory connectionFactory;
    private final OpenAppService openAppService;
    private ReactiveRedisMessageListenerContainer container;
    private Disposable subscription;

    // 构造器注入两者

    @Override
    public void start() {
        container = new ReactiveRedisMessageListenerContainer(connectionFactory);
        subscription = container.receive(ChannelTopic.of(CHANNEL))
                .map(ReactiveSubscription.Message::getMessage)
                .doOnNext(appKey -> {
                    if (appKey == null || appKey.isEmpty()) {
                        openAppService.invalidateAll();
                    } else {
                        openAppService.invalidate(appKey);
                    }
                })
                .onErrorContinue((e, v) -> log.warn("[start] 处理应用缓存失效消息失败", e))
                .subscribe();

        if (subscription != null) {
            subscription.dispose();
        }
        if (container != null) {
            container.destroyLater().subscribe();
        }
    }

    public boolean isRunning() {
        return subscription != null && !subscription.isDisposed();
    }
}
```

```java
package com.ccloud.module.open.support;

/**
 * 测试用回调接收方：记录收到的请求，按 status 应答；redirectTo 不为空时返回 302
 */
public final class NotifyReceiver implements AutoCloseable {

    public record Received(Headers headers, String body) {
    }

    public final List<Received> requests = new CopyOnWriteArrayList<>();
    public volatile int status = 200;
    public volatile String responseBody = "success";
    public volatile String redirectTo;
    public volatile long delayMillis;

    private final HttpServer server;

    public NotifyReceiver() throws IOException {
        server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
        server.createContext("/", exchange -> {
            Headers headers = new Headers();
            headers.putAll(exchange.getRequestHeaders());
            requests.add(new Received(headers, new String(exchange.getRequestBody().readAllBytes(), UTF_8)));
            if (delayMillis > 0) {
                try { Thread.sleep(delayMillis); } catch (InterruptedException ignored) { Thread.currentThread().interrupt(); }
            }
            if (redirectTo != null) {
                exchange.getResponseHeaders().add("Location", redirectTo);
                exchange.sendResponseHeaders(302, -1);
                exchange.close();
                return;
            }
            byte[] bytes = responseBody.getBytes(UTF_8);
            exchange.sendResponseHeaders(status, bytes.length);
            try (OutputStream out = exchange.getResponseBody()) {
                out.write(bytes);
            }
        });
        server.start();
    }

    public String url(String path) {
        return "http://127.0.0.1:" + server.getAddress().getPort() + path;
    }

    @Override
    public void close() {
        server.stop(0);
    }
}
```

```java
package com.ccloud.module.open.framework.notify;

/**
 * 只放行公网地址的 DNS 解析器。解析结果即连接地址，校验与连接用的是同一组地址，DNS 重绑定无法绕过。
 * 任何一个解析结果受限即整体拒绝：只挑合法地址连接，攻击者仍可让部分请求打到内网
 */
public class RestrictedDnsResolver implements DnsResolver {

    private final boolean allowPrivate;

    public RestrictedDnsResolver(boolean allowPrivate) {
        this.allowPrivate = allowPrivate;
    }

    @Override
    public InetAddress[] resolve(String host) throws UnknownHostException {
        InetAddress[] addresses = SystemDefaultDnsResolver.INSTANCE.resolve(host);
        for (InetAddress address : addresses) {
            check(address, allowPrivate);
        }
        return addresses;
    }

    @Override
    public String resolveCanonicalHostname(String host) throws UnknownHostException {
        return SystemDefaultDnsResolver.INSTANCE.resolveCanonicalHostname(host);
    }

    static void check(InetAddress address, boolean allowPrivate) throws UnknownHostException {
        if (isMetadata(address) || (!allowPrivate && isPrivate(address))) {
            throw new UnknownHostException("回调地址解析到受限地址：" + address.getHostAddress());
        }
    }

    private static boolean isMetadata(InetAddress address) {
        String ip = address.getHostAddress();
        return "169.254.169.254".equals(ip) || ip.startsWith("fd00:ec2:0:0:0:0:0:254");
    }

    private static boolean isPrivate(InetAddress address) {
        if (address.isAnyLocalAddress() || address.isLoopbackAddress() || address.isLinkLocalAddress()
                || address.isSiteLocalAddress() || address.isMulticastAddress()) {
            return true;
        }
        byte[] b = address.getAddress();
        if (b.length == 4) {
            int first = b[0] & 0xFF;
            int second = b[1] & 0xFF;
            return (first == 100 && second >= 64 && second <= 127)   // 100.64.0.0/10 运营商级 NAT
                    || first == 0 || first >= 240;                   // 0.0.0.0/8、保留地址
        }
        return (b[0] & 0xFE) == 0xFC;                                // fc00::/7 唯一本地地址
    }
}
```

```java
package com.ccloud.module.open.framework.notify;

/**
 * 回调的 HTTP 发送。不跟随重定向（302 到内网地址可绕过地址校验）、不自动重试（重试由回调日志的退避策略负责）
 */
@Slf4j
@Component
public class OpenNotifyClient implements DisposableBean {

    static final int MAX_RESPONSE_LENGTH = 2048;

    private final boolean allowPrivate;
    private final CloseableHttpClient httpClient;

    @Autowired
    public OpenNotifyClient(OpenProperties openProperties) {
        this(openProperties.getNotify());
    }

    OpenNotifyClient(OpenProperties.Notify notify) {
        this.allowPrivate = notify.isAllowPrivateAddress();
        PoolingHttpClientConnectionManager connectionManager = PoolingHttpClientConnectionManagerBuilder.create()
                .setDnsResolver(new RestrictedDnsResolver(allowPrivate))
                .setDefaultConnectionConfig(ConnectionConfig.custom()
                        .setConnectTimeout(Timeout.of(notify.getConnectTimeout()))
                        .setSocketTimeout(Timeout.of(notify.getReadTimeout()))
                        .build())
                .setMaxConnTotal(50)
                .setMaxConnPerRoute(5)
                .build();
        this.httpClient = HttpClients.custom()
                .setConnectionManager(connectionManager)
                .disableRedirectHandling()
                .disableAutomaticRetries()
                .disableCookieManagement()
                .setDefaultRequestConfig(RequestConfig.custom()
                        .setResponseTimeout(Timeout.of(notify.getReadTimeout()))
                        .build())
                .build();
    }

    public NotifyResult post(String url, Map<String, String> headers, String body) {
        URI uri;
        try {
            uri = new URI(url);
        } catch (URISyntaxException e) {
            return new NotifyResult(false, 0, "回调地址不合法");
        }
        if (!"http".equalsIgnoreCase(uri.getScheme()) && !"https".equalsIgnoreCase(uri.getScheme())
                || uri.getHost() == null) {
            return new NotifyResult(false, 0, "回调地址须为 http 或 https");
        }
        // 字面量 IP 不一定经过 DNS 解析器，这里先校验一次
        String host = uri.getHost().replaceAll("^\\[|]$", "");
        if (InetAddressUtils.isIPv4Address(host) || InetAddressUtils.isIPv6Address(host)) {
            try {
                RestrictedDnsResolver.check(InetAddress.getByName(host), allowPrivate);
            } catch (UnknownHostException e) {
                return new NotifyResult(false, 0, e.getMessage());
            }
        }
        HttpPost post = new HttpPost(uri);
        headers.forEach(post::setHeader);
        post.setEntity(new StringEntity(body, ContentType.APPLICATION_JSON));
        try {
            return httpClient.execute(post, response -> {
                int status = response.getCode();
                String text = response.getEntity() == null ? "" : readLimited(response.getEntity().getContent());
                return new NotifyResult(status >= 200 && status < 300, status, text);
            });
        } catch (UnknownHostException e) {
            return new NotifyResult(false, 0, StrUtil.sub(e.getMessage(), 0, MAX_RESPONSE_LENGTH));
        } catch (IOException e) {
            log.info("[post] 回调发送失败 url={}：{}", url, e.toString());
            return new NotifyResult(false, 0, StrUtil.sub("发送失败：" + e.getMessage(), 0, MAX_RESPONSE_LENGTH));
        }
    }

    private static String readLimited(InputStream in) throws IOException {
        byte[] bytes = in.readNBytes(MAX_RESPONSE_LENGTH * 4);
        return StrUtil.sub(new String(bytes, StandardCharsets.UTF_8), 0, MAX_RESPONSE_LENGTH);
    }

    @Override
    public void destroy() throws IOException {
        httpClient.close();
    }
}
```

```java
@Slf4j
@Service
public class OpenNotifyServiceImpl implements OpenNotifyService, DisposableBean {

    static final Duration[] BACKOFF = {Duration.ofSeconds(15), Duration.ofMinutes(1), Duration.ofMinutes(5),
            Duration.ofMinutes(30), Duration.ofHours(2), Duration.ofHours(6)};
    /**
     * 抢占后的租约：发送最长约 10 秒（连接 + 读取超时），租约内其他执行方不会重复发送；进程在发送中崩溃时租约到期后由定时任务接手
     */
    static final Duration LEASE = Duration.ofSeconds(60);

    @Resource
    private OpenNotifyLogMapper openNotifyLogMapper;
    @Resource
    private OpenAppService openAppService;
    @Resource
    private OpenNotifyClient openNotifyClient;

    /**
     * 即时发送用的线程池。队列满时丢弃：记录已落库，定时任务会接手
     */
    private final ThreadPoolExecutor executor = new ThreadPoolExecutor(2, 4, 60, TimeUnit.SECONDS,
            new ArrayBlockingQueue<>(1000), r -> {
                Thread thread = new Thread(r, "open-notify");
                thread.setDaemon(true);
                return thread;
            }, (r, e) -> log.warn("[open-notify] 即时发送队列已满，交由定时任务发送"));

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Long sendNotify(Long appId, String event, Map<String, Object> data) {
        OpenAppDO app = openAppService.getAppIgnoreTenant(appId);
        if (app == null) {
            throw exception(OPEN_APP_NOT_EXISTS);
        }
        String eventId = IdUtil.fastSimpleUUID();
        Map<String, Object> body = new LinkedHashMap<>();
        body.put("eventId", eventId);
        body.put("event", event);
        body.put("appKey", app.getAppKey());
        body.put("timestamp", System.currentTimeMillis());
        body.put("data", data);
        boolean noUrl = StrUtil.isBlank(app.getNotifyUrl());
        OpenNotifyLogDO notifyLog = new OpenNotifyLogDO().setAppId(appId).setEventId(eventId).setEvent(event)
                .setPayload(JsonUtils.toJsonString(body))
                .setNotifyUrl(StrUtil.nullToEmpty(app.getNotifyUrl()))
                .setStatus(noUrl ? OpenNotifyStatusEnum.ABANDONED.getStatus() : OpenNotifyStatusEnum.WAITING.getStatus())
                .setRetryCount(0)
                .setNextRetryTime(noUrl ? null : LocalDateTime.now())
                .setResponseBody(noUrl ? "应用未配置回调地址" : null);
        // 回调记录属于应用所在租户；调用方可能是其他租户上下文中的业务服务
        notifyLog.setTenantId(app.getTenantId());
        TenantUtils.executeIgnore(() -> openNotifyLogMapper.insert(notifyLog));
        if (!noUrl) {
            executeAfterCommit(notifyLog.getId());
        }
        return notifyLog.getId();
    }

    @Override
    public Long sendTestNotify(Long appId) {
        OpenAppDO app = openAppService.getApp(appId);
        if (app == null) {
            throw exception(OPEN_APP_NOT_EXISTS);
        }
        if (StrUtil.isBlank(app.getNotifyUrl())) {
            throw exception(OPEN_APP_NOTIFY_URL_NOT_CONFIGURED);
        }
        return sendNotify(appId, OpenNotifyEvents.TEST, Map.of("message", "ccloud test"));
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void retryNotify(Long id) {
        OpenNotifyLogDO notifyLog = getNotifyLog(id);
        if (notifyLog == null) {
            throw exception(OPEN_NOTIFY_LOG_NOT_EXISTS);
        }
        if (!OpenNotifyStatusEnum.FAILURE.getStatus().equals(notifyLog.getStatus())
                && !OpenNotifyStatusEnum.ABANDONED.getStatus().equals(notifyLog.getStatus())) {
            throw exception(OPEN_NOTIFY_LOG_CANNOT_RETRY);
        }
        openNotifyLogMapper.updateById(new OpenNotifyLogDO().setId(id)
                .setStatus(OpenNotifyStatusEnum.WAITING.getStatus()).setRetryCount(0)
                .setNextRetryTime(LocalDateTime.now()));
        executeAfterCommit(id);
    }

    @Override
    public int executeDueNotifies(int limit) {
        List<Long> ids = TenantUtils.executeIgnore(() -> openNotifyLogMapper.selectDueIds(LocalDateTime.now(), limit));
        int count = 0;
        for (Long id : ids) {
            try {
                if (execute(id)) {
                    count++;
                }
            } catch (RuntimeException e) {
                // 单条异常不影响其他记录；租约到期后会再次被扫到
                log.error("[executeDueNotifies] 回调 {} 处理异常", id, e);
            }
        }
        return count;
    }

    /**
     * 抢占并发送一次，按结果更新状态。返回是否抢占成功
     */
    boolean execute(Long id) {
        LocalDateTime now = LocalDateTime.now();
        if (TenantUtils.executeIgnore(() -> openNotifyLogMapper.claim(id, now, now.plus(LEASE))) == 0) {
            return false;
        }
        OpenNotifyLogDO notifyLog = TenantUtils.executeIgnore(() -> openNotifyLogMapper.selectById(id));
        OpenAppDO app = openAppService.getAppIgnoreTenant(notifyLog.getAppId());
        OpenNotifyLogDO update = new OpenNotifyLogDO().setId(id);
        NotifyResult result;
        if (app == null || StrUtil.isBlank(app.getNotifyUrl())) {
            result = new NotifyResult(false, 0, app == null ? "应用已删除" : "应用未配置回调地址");
            update.setStatus(OpenNotifyStatusEnum.ABANDONED.getStatus());
        } else {
            // 重试时用应用当前的地址与密钥：管理员修正回调地址、重置密钥后，后续重试随之生效
            long timestamp = System.currentTimeMillis();
            Map<String, String> headers = Map.of(
                    "X-Open-App-Id", app.getAppKey(),
                    "X-Open-Timestamp", String.valueOf(timestamp),
                    "X-Open-Event-Id", notifyLog.getEventId(),
                    "X-Open-Sign", OpenNotifySigner.sign(app.getAppSecret(), timestamp, notifyLog.getEventId(),
                            notifyLog.getPayload()));
            update.setNotifyUrl(app.getNotifyUrl());
            result = openNotifyClient.post(app.getNotifyUrl(), headers, notifyLog.getPayload());
            if (result.success()) {
                update.setStatus(OpenNotifyStatusEnum.SUCCESS.getStatus());
            } else {
                int retried = notifyLog.getRetryCount();
                if (retried < BACKOFF.length) {
                    update.setStatus(OpenNotifyStatusEnum.FAILURE.getStatus()).setRetryCount(retried + 1)
                            .setNextRetryTime(LocalDateTime.now().plus(BACKOFF[retried]));
                } else {
                    update.setStatus(OpenNotifyStatusEnum.ABANDONED.getStatus());
                }
            }
        }
        update.setHttpStatus(result.httpStatus()).setResponseBody(result.responseBody());
        TenantUtils.executeIgnore(() -> openNotifyLogMapper.updateById(update));
        return true;
    }

    private void executeAfterCommit(Long id) {
        Runnable task = () -> executor.execute(() -> {
            try {
                execute(id);
            } catch (RuntimeException e) {
                log.error("[executeAfterCommit] 回调 {} 即时发送异常，交由定时任务重试", id, e);
            }
        });
        if (TransactionSynchronizationManager.isSynchronizationActive()) {
            TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
                @Override
                public void afterCommit() {
                    task.run();
                }
            });
        } else {
            task.run();
        }
    }

    @Override
    public void destroy() throws InterruptedException {
```
